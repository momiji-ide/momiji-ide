interface Planet {
  name: string
  radius: number
  distance: number
  color: string
}

const createOrbit = (planet: Planet): void => {
  const canvas = document.getElementById('canvas')
  const ctx = canvas?.getContext('2d')
  if (!ctx) return

  ctx.beginPath()
  ctx.arc(400, 300, planet.distance, 0, Math.PI * 2)
  ctx.strokeStyle = planet.color
  ctx.stroke()
}
